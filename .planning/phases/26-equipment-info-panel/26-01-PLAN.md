<objective>
Establish the equipment-specs module (EquipmentSpec type, EquipmentEfficiencyGrade, Korean grade tables, inference rules) and extend the selection-store with a serialisable selectedEquipment slice. This is the pure data foundation Plan 02 consumes when wiring the click handler and info panel.

Purpose: The foundation must exist before any UI or scene code can safely extract userData from a clicked MEP mesh. Co-locating types, grade tables, and inference in one pure module (no React, no THREE.js) lets the click handler stay trivial — it just calls `inferEquipmentSpecs(userData, recipe)` and stores the resulting plain object.

Output: Typed EquipmentSpec + SelectedEquipmentInfo contracts, era-based inference tables, and the store extension — all backed by Vitest tests.

    // src/lib/energy/equipment-specs.ts
    import type { BuildingEra } from "@/lib/material-types";
    import type { BuildingRecipe } from "@/lib/procedural/types";

    /** KS B 6364 (HVAC) / KSC IEC 62301 (electrical) equipment efficiency grade.
     *  Scale: 1 (best, 우수) to 5 (worst, 불량).
     *  SEPARATE from building-level EnergyGrade (1+++ to 7) — do NOT import EnergyGrade here. */
    export type EquipmentEfficiencyGrade = 1 | 2 | 3 | 4 | 5;

    export type EquipmentDataSource = "estimated-from-era" | "estimated-from-recipe";

    export interface EquipmentSpec {
      categoryKo: string;
      categoryEn: string;
      capacity: string;           // e.g. "12 kW" or "500 W/fixture"
      installYear: number;        // era midpoint from ERA_INSTALL_YEAR
      annualKwh: number;
      efficiencyGrade: EquipmentEfficiencyGrade;
      efficiencyGradeLabel: string;  // e.g. "1등급 (우수)"
      gradeColor: string;         // hex color from EQUIPMENT_GRADE_COLORS
      dataSource: EquipmentDataSource;
      /** Korean standard reference for the grade */
      standardRef: "KS B 6364" | "KSC IEC 62301";
    }

    export const ERA_INSTALL_YEAR: Record<BuildingEra, number> = {
      "pre-1970":  1965,
      "1970-1989": 1979,
      "1990-1999": 1994,
      "2000-2009": 2004,
      "2010-2019": 2014,
      "2020+":     2022,
    };

    export const HVAC_ERA_GRADE: Record<BuildingEra, EquipmentEfficiencyGrade> = {
      "pre-1970":  5,
      "1970-1989": 5,
      "1990-1999": 4,
      "2000-2009": 3,
      "2010-2019": 2,
      "2020+":     1,
    };

    export const ELECTRICAL_ERA_GRADE: Record<BuildingEra, EquipmentEfficiencyGrade> = {
      "pre-1970":  5,
      "1970-1989": 5,
      "1990-1999": 4,
      "2000-2009": 3,
      "2010-2019": 2,
      "2020+":     1,
    };

      1: "1등급 (우수)",
      2: "2등급 (양호)",
      3: "3등급 (보통)",
      4: "4등급 (미흡)",
      5: "5등급 (불량)",
    };

    export const EQUIPMENT_GRADE_COLORS: Record<EquipmentEfficiencyGrade, string> = {
      1: "#16a34a",  // green (best)
      2: "#84cc16",  // lime
      3: "#facc15",  // yellow
      4: "#f97316",  // orange
      5: "#dc2626",  // red (worst)
    };

    /** Pure, synchronous inference from userData + recipe.
     *  No React, no THREE.js, no store reads, no async. */
    export function inferEquipmentSpecs(
      userData: { type?: string; floorNo?: number | null },
      recipe: BuildingRecipe
    ): EquipmentSpec {
      const type = userData.type ?? "unknown";
      const era = recipe.era;
      const installYear = ERA_INSTALL_YEAR[era];
      const floorArea = Math.max(recipe.footprintWidth * recipe.footprintDepth, 1);
      const floorCount = Math.max(recipe.floors.length, 1);

      // Dispatch on userData.type PREFIX (cooling-*, heating-*, etc.)
      // Derive HVAC vs electrical grade from tables above.
      // Capacity + annualKwh derived per RESEARCH.md §Pattern 5:
      //   - HVAC branch: capacity ~= 0.05 kW/m²·floor
      //   - Lighting fixture: capacity ~= 10 W/m², annual = W/m² × area × 2500h (office default)
      //   - DHW: 15% of total electrical ~5 kWh/m²·yr × area
      //   - Electrical shell/microgrid: residual
      // Concrete lookup per-prefix — see RESEARCH.md §Pattern 5 for the exact rules.
      // If type prefix is unknown, return a default spec with categoryKo="기타", grade=3,
      //   standardRef="KSC IEC 62301", dataSource="estimated-from-recipe".
      // (Implementation: use a switch on the prefix before the first "-".)
      ...
    }
