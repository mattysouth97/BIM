    **Create `src/lib/workflow/stages.ts`:**

    ```typescript
    export type WorkflowStage = "select" | "assemble" | "configure" | "analyze" | "export";

    export const STAGE_ORDER: WorkflowStage[] = [
      "select", "assemble", "configure", "analyze", "export"
    ];

    export const STAGE_LABELS: Record<WorkflowStage, { ko: string; en: string }> = {
      select:    { ko: "건물 선택",   en: "Select Building" },
      assemble:  { ko: "구조 조립",   en: "Assemble" },
      configure: { ko: "속성 설정",   en: "Configure" },
      analyze:   { ko: "분석",       en: "Analyze" },
      export:    { ko: "내보내기",    en: "Export" },
    };

    // DAG prerequisite guards — pure functions.
    // Return true if the user CAN leave this stage (advance forward).
    // Guards are intentionally permissive in v3.0 — stages are suggested, not enforced.
    export const STAGE_GUARDS: Partial<Record<WorkflowStage, () => boolean>> = {
      select:    () => true,
      assemble:  () => true,
      configure: () => true,
      analyze:   () => true,
      // "export" has no forward guard — it is the terminal stage.
    };
    ```

    **Create `src/store/workflow-store.ts`:**

    Use `"use client"` directive. Follow the exact Zustand persist pattern from app-store.ts:
    - `create<WorkflowState>()(persist((set, get) => ({ ... }), { name: "bim-workflow-state", partialize: ... }))`
    - partialize returns only `{ stage, completion }` — no functions serialized

    State shape:
    ```typescript
    interface WorkflowState {
      stage: WorkflowStage;
      completion: Record<WorkflowStage, boolean>;
      setStage: (next: WorkflowStage) => void;
      canAdvance: () => boolean;
      advance: () => void;
      retreat: () => void;
      markComplete: (stage: WorkflowStage) => void;
      resetWorkflow: () => void;
    }
    ```

    Implementation details:
    - `advance()`: get current index in STAGE_ORDER, if < length-1, set stage to STAGE_ORDER[idx+1]. If at terminal, no-op.
    - `retreat()`: get current index, if > 0, set stage to STAGE_ORDER[idx-1]. If at start, no-op.
    - `canAdvance()`: look up STAGE_GUARDS[current stage], call it. If no guard exists (terminal), return false.
    - `markComplete(s)`: set completion[s] = true without touching other flags.
    - `resetWorkflow()`: set stage to "select", completion to all false.
