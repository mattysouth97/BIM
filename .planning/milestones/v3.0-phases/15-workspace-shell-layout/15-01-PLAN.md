```typescript
"use client";

import { Button } from "@/components/ui/button";
import { ChevronLeft, ChevronRight } from "lucide-react";
import { cn } from "@/lib/utils";

interface DockCollapseButtonProps {
  side: "left" | "right";
  collapsed: boolean;
  onClick: () => void;
  className?: string;
}

export function DockCollapseButton({ side, collapsed, onClick, className }: DockCollapseButtonProps) {
  // When left dock is open, show "<<" to collapse (pointing left = hide left)
  // When left dock is collapsed, show ">>" to expand (pointing right = show left)
  // When right dock is open, show ">>" to collapse (pointing right = hide right)
  // When right dock is collapsed, show "<<" to expand (pointing left = show right)
  const showChevronLeft = (side === "left" && !collapsed) || (side === "right" && collapsed);

  return (
    <Button
      variant="ghost"
      size="icon"
      className={cn("h-6 w-6 shrink-0", className)}
      onClick={onClick}
      title={collapsed ? "Expand panel" : "Collapse panel"}
    >
      {showChevronLeft ? (
        <ChevronLeft className="size-4" />
      ) : (
        <ChevronRight className="size-4" />
      )}
    </Button>
  );
}
```

2. Create `src/components/workspace/workspace-shell.tsx` — the root layout component:

Props: `children` (React node rendered in the center viewport panel), plus any data props needed by future dock content.

Structure (per D-01):
- Outer: `div.flex.flex-col.h-full` (fills parent height)
- Inner: `ResizablePanelGroup direction="horizontal" className="flex-1 min-h-0"`
  - Left dock: `ResizablePanel` with `defaultSize={LEFT_DOCK_DEFAULT}` `minSize={LEFT_DOCK_MIN}` `maxSize={LEFT_DOCK_MAX}`. **ALWAYS rendered** — use the `hidden` prop (`hidden={!leftDockOpen}`) or apply `style={{ display: leftDockOpen ? undefined : 'none' }}` to hide it when collapsed. Do NOT use conditional JSX (`{leftDockOpen && <ResizablePanel>}`). Conditional rendering causes the PanelGroup to re-measure and animate a layout jump, and loses persisted size on re-expand.
  - `ResizableHandle withHandle` between left and center — also hidden when left dock is collapsed (use same `hidden` or `display:none` pattern on the handle)
  - Center viewport: `ResizablePanel` with no min/max constraints (takes remaining space), renders `children`
  - `ResizableHandle withHandle` between center and right — also hidden when right dock is collapsed
  - Right dock: `ResizablePanel` with `defaultSize={RIGHT_DOCK_DEFAULT}` `minSize={RIGHT_DOCK_MIN}` `maxSize={RIGHT_DOCK_MAX}`. **ALWAYS rendered** — use the `hidden` prop or `display:none` when collapsed, same as left dock. Do NOT conditionally render.
- Bottom shelf (per D-06): A collapsible `div` below the ResizablePanelGroup, visible when `bottomShelfOpen`. Height fixed at `h-10` placeholder. Contains empty placeholder text "Status bar (Phase 18)".

Key implementation details:
- Read `leftDockOpen`, `rightDockOpen`, `bottomShelfOpen` from `useWorkspaceStore` via individual selectors (not a single object selector — avoid unnecessary re-renders per PITFALLS.md)
- Use `useHydration()` hook before reading workspace-store to avoid SSR mismatch (per CLAUDE.md known issue)
- Persist panel sizes: Use `onLayoutChanged` callback on ResizablePanelGroup. This fires ONCE after drag completes with the final sizes array. Call `setLeftDockSize` / `setRightDockSize` with the appropriate values from the sizes array. Do NOT use `onLayout` (fires on every pointer-move during drag, causing 60 re-renders/second — see PITFALLS.md Pitfall 7).
- Left dock content: render a `div` with a header containing `DockCollapseButton` on the right edge + placeholder text "Left dock (Phase 17)"
- Right dock content: render a `div` with a header containing `DockCollapseButton` on the left edge + placeholder text "Right dock (Phase 17)"
- Collapse buttons for hidden docks: When a dock is collapsed, render a narrow `DockCollapseButton` at the edge of the viewport panel so the user can re-expand it
- Use `pointer-events-auto` on dock panels, let center viewport get events normally (per D-07)
- The center viewport panel should have `className="relative"` so children (BuildingScene) can position absolutely inside it
- Use `"use client"` directive
