    Polygon buildings have N edges instead of 4 faces. Each consecutive pair of outer ring
    vertices defines one facade strip (same role as a "face" in getFaces()).

    Add this function in facade-generator.ts:

    ```typescript
    /**
     * Derive FaceDesc array from cadastral polygon outer ring edges.
     * Each consecutive vertex pair becomes one facade face strip.
     * Used instead of getFaces() when recipe.footprintPolygon is present.
     */
    function getPolygonFaces(outerRing: [number, number][], wallThickness: number): FaceDesc[] {
      const faces: FaceDesc[] = [];
      const n = outerRing.length;

      for (let i = 0; i < n - 1; i++) {  // n-1 because last point closes ring (equals first)
        const [x0, z0] = outerRing[i];
        const [x1, z1] = outerRing[i + 1];

        const dx = x1 - x0;
        const dz = z1 - z0;
        const edgeLength = Math.sqrt(dx * dx + dz * dz);
        if (edgeLength < 0.1) continue;  // skip degenerate edges

        // Midpoint of this edge
        const midX = (x0 + x1) / 2;
        const midZ = (z0 + z1) / 2;

        // Edge direction angle (atan2 in XZ plane)
        const angle = Math.atan2(dx, dz);  // angle from +Z axis toward +X axis

        // Position the face at the outer wall surface
        // Offset inward by wallThickness/2 from the midpoint along the face normal
        const nx = -dz / edgeLength;  // outward normal X (perpendicular to edge, pointing out)
        const nz = dx / edgeLength;   // outward normal Z

        const facePos = new THREE.Vector3(
          midX - nx * wallThickness / 2,
          0,
          midZ - nz * wallThickness / 2,
        );

        // Quaternion: rotate a +Z-facing quad to align with this edge
        const quat = new THREE.Quaternion().setFromAxisAngle(
          new THREE.Vector3(0, 1, 0),
          -angle,
        );

        faces.push({
          position: facePos,
          quaternion: quat,
          length: edgeLength,
          side: "front",  // polygon faces are all treated as front for window-ratio purposes
        });
      }

      return faces;
    }
    ```
