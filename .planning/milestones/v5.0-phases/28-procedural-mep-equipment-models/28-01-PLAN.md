// src/store/material-store.ts shows the overrideProperty nested-path pattern this plan MUST mirror:
overrideProperty: (pk, path, value) =>
  set((state) => {
    const current = state.properties[pk];
    if (!current) return state;
    const updated = JSON.parse(JSON.stringify(current));
    const parts = path.split(".");
    let obj = updated;
    for (let i = 0; i < parts.length - 1; i++) {
      obj = obj[parts[i]];
    }
    obj[parts[parts.length - 1]] = value;
    return { properties: { ...state.properties, [pk]: updated } };
  });

// equipment-store.overrideParam differs in two ways:
// 1. If params[pk] is absent, INITIALIZE from DEFAULT_MEP_EQUIPMENT_PARAMS then override
//    (material-store returns state unchanged — that would silently drop user input in our case)
// 2. No `source` field to set (equipment params are pure geometry, not thermal data)
</reference_pattern>
