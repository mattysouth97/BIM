// From 28-RESEARCH.md lines 499–522 — canonical buildChillerGeometry:
import { mergeGeometries } from "three/examples/jsm/utils/BufferGeometryUtils.js";

function buildChillerGeometry(p: ChillerParams): THREE.BufferGeometry {
  const body = new THREE.BoxGeometry(p.bodyWidth, p.bodyHeight, p.bodyDepth);
  const grille = new THREE.BoxGeometry(p.bodyWidth * 0.9, p.bodyHeight * 0.9, 0.08);
  grille.translate(0, 0, p.bodyDepth / 2 + 0.04);
  const pipeA = new THREE.CylinderGeometry(p.pipeStubRadius, p.pipeStubRadius, 0.4, 8);
  pipeA.rotateZ(Math.PI / 2);
  pipeA.translate(p.bodyWidth / 2 + 0.2, -p.bodyHeight * 0.3, 0);
  const pipeB = new THREE.CylinderGeometry(p.pipeStubRadius * 0.8, p.pipeStubRadius * 0.8, 0.4, 8);
  pipeB.rotateZ(Math.PI / 2);
  pipeB.translate(p.bodyWidth / 2 + 0.2, p.bodyHeight * 0.3, 0);
  return mergeGeometries([body, grille, pipeA, pipeB]);
}

// Canonical boiler — CylinderGeometry body + Cylinder flue + 2 pipe stubs:
function buildBoilerGeometry(p: BoilerParams): THREE.BufferGeometry {
  const body = new THREE.CylinderGeometry(p.radius, p.radius, p.height, 16);
  const flue = new THREE.CylinderGeometry(p.flueRadius, p.flueRadius, p.flueHeight, 12);
  flue.translate(0, p.height / 2 + p.flueHeight / 2, 0);
  const pipeA = new THREE.CylinderGeometry(0.08, 0.08, 0.5, 8);
  pipeA.rotateZ(Math.PI / 2);
  pipeA.translate(p.radius + 0.25, -p.height * 0.3, 0);
  const pipeB = new THREE.CylinderGeometry(0.08, 0.08, 0.5, 8);
  pipeB.rotateZ(Math.PI / 2);
  pipeB.translate(p.radius + 0.25, p.height * 0.3, 0);
  return mergeGeometries([body, flue, pipeA, pipeB]);
}
